cmake_minimum_required(VERSION 3.24)
project(CopypartyViewer VERSION 2.0.0 LANGUAGES C CXX)

option(CPV_SUPERBUILD "Build the pinned private static dependencies first" ON)
option(CPV_BUILD_TESTS "Build meaningful core and preview regression tests" ON)
option(CPV_VERIFY_PUBLIC_DEFAULTS "Test the public release's embedded defaults" OFF)
option(CPV_TEST_HTTP "Run the local HTTP fixture tests (requires Python 3)" OFF)
option(CPV_TEST_RETRO "Build and run Retro widget tests (requires a display)" OFF)
set(CPV_DEFAULT_URL "https://a.ocv.me/pub/" CACHE STRING "First-run server URL")
set(CPV_DEFAULT_USERNAME "" CACHE STRING "Optional first-run username; blank in official releases")
set(CPV_DEFAULT_PASSWORD "" CACHE STRING "Optional first-run password; blank in official releases")
set(CPV_DEPS_PREFIX "" CACHE PATH "Private dependency installation prefix")
set(CPV_FLTK_SOURCE_DIR "" CACHE PATH "Pinned FLTK source directory (NanoSVG headers)")
set(CPV_JOBS "4" CACHE STRING "Parallel jobs per dependency")
set(CPV_SITE_CONFIG "" CACHE FILEPATH "Optional private CMake file setting first-run defaults")
if(CPV_SITE_CONFIG)
  include("${CPV_SITE_CONFIG}")
endif()

# Keep diagnostic __FILE__ strings independent of the builder's local paths.
# The flags are passed through the superbuild to private dependencies as well.
if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
  foreach(language C CXX)
    foreach(mapping "${CMAKE_SOURCE_DIR}=." "${CMAKE_BINARY_DIR}=./build")
      set(prefix_flag "\"-ffile-prefix-map=${mapping}\"")
      string(FIND "${CMAKE_${language}_FLAGS}" "${prefix_flag}" found)
      if(found EQUAL -1)
        string(APPEND CMAKE_${language}_FLAGS " ${prefix_flag}")
      endif()
    endforeach()
  endforeach()
endif()

if(CPV_SUPERBUILD)
  include(cmake/Superbuild.cmake)
  return()
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_SKIP_RPATH TRUE)
if(NOT CPV_DEPS_PREFIX OR NOT CPV_FLTK_SOURCE_DIR)
  message(FATAL_ERROR "Use the top-level Makefile or supply the private dependency prefix and FLTK source directory")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${CPV_DEPS_PREFIX}")
find_package(Threads REQUIRED)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED PATHS "${CPV_DEPS_PREFIX}" NO_DEFAULT_PATH)
foreach(bundled_codec fltk::png fltk::jpeg fltk::z)
  if(TARGET ${bundled_codec})
    message(FATAL_ERROR "FLTK fell back to its bundled codec ${bundled_codec}; the private pinned PNG/JPEG/zlib libraries are required")
  endif()
endforeach()
find_package(CURL 8.23.0 EXACT CONFIG REQUIRED PATHS "${CPV_DEPS_PREFIX}" NO_DEFAULT_PATH)
find_package(PNG 1.6.59 EXACT REQUIRED)
find_package(JPEG REQUIRED)
find_package(ZLIB 1.4 EXACT REQUIRED)
find_library(CPV_WEBP_LIBRARY NAMES webpdecoder PATHS "${CPV_DEPS_PREFIX}/lib" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)
find_library(CPV_WEBPDEMUX_LIBRARY NAMES webpdemux PATHS "${CPV_DEPS_PREFIX}/lib" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
find_path(CPV_JSON_INCLUDE nlohmann/json.hpp PATHS "${CPV_DEPS_PREFIX}/include" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH REQUIRED)

add_library(cpv_core STATIC src/core.cpp)
target_include_directories(cpv_core PUBLIC src "${CPV_JSON_INCLUDE}")
target_compile_definitions(cpv_core PUBLIC CURL_STATICLIB)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${CMAKE_BINARY_DIR}/generated/cpv_defaults.hpp" "// Generated build defaults. Do not distribute a private credential build.\n#pragma once\n")
foreach(key URL USERNAME PASSWORD)
  string(REPLACE "\\" "\\\\" escaped "${CPV_DEFAULT_${key}}")
  string(REPLACE "\"" "\\\"" escaped "${escaped}")
  string(REPLACE "\r" "\\r" escaped "${escaped}")
  string(REPLACE "\n" "\\n" escaped "${escaped}")
  string(REPLACE "\t" "\\t" escaped "${escaped}")
  file(APPEND "${CMAKE_BINARY_DIR}/generated/cpv_defaults.hpp" "#define CPV_DEFAULT_${key} \"${escaped}\"\n")
endforeach()
target_compile_options(cpv_core PRIVATE -include "${CMAKE_BINARY_DIR}/generated/cpv_defaults.hpp")
target_link_libraries(cpv_core PUBLIC CURL::libcurl Threads::Threads)
if(WIN32)
  target_link_libraries(cpv_core PUBLIC shlwapi shell32 ole32 advapi32 bcrypt)
endif()

add_library(cpv_preview STATIC src/preview.cpp src/text_encoding.cpp)
target_include_directories(cpv_preview PUBLIC src PRIVATE "${CPV_FLTK_SOURCE_DIR}/nanosvg" "${CPV_DEPS_PREFIX}/include")
target_link_libraries(cpv_preview PUBLIC cpv_core fltk::images fltk::fltk PNG::PNG JPEG::JPEG ZLIB::ZLIB "${CPV_WEBP_LIBRARY}")
if(CPV_WEBPDEMUX_LIBRARY)
  target_link_libraries(cpv_preview PUBLIC "${CPV_WEBPDEMUX_LIBRARY}")
endif()

# Small native document parsers and code lexer; no browser runtime is required.
add_library(cpv_document STATIC src/document.cpp src/rtf.cpp src/csv.cpp vendor/md4c/md4c.c vendor/md4c/entity.c)
target_include_directories(cpv_document PUBLIC src PRIVATE vendor/md4c)
set_target_properties(cpv_document PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)

if(WIN32)
  enable_language(RC)
  add_executable(copyparty-viewer WIN32 src/main.cpp src/ui.cpp src/platform.cpp src/retro.cpp packaging/windows.rc)
  set_target_properties(copyparty-viewer PROPERTIES OUTPUT_NAME CopypartyViewer)
  target_link_libraries(copyparty-viewer PRIVATE dwmapi uxtheme comctl32)
else()
  add_executable(copyparty-viewer src/main.cpp src/ui.cpp src/platform.cpp src/retro.cpp)
endif()
target_include_directories(copyparty-viewer PRIVATE src "${CMAKE_CURRENT_SOURCE_DIR}/assets")
target_link_libraries(copyparty-viewer PRIVATE cpv_preview cpv_document ${CMAKE_DL_LIBS})

if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  foreach(target cpv_core cpv_preview cpv_document copyparty-viewer)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffunction-sections -fdata-sections)
  endforeach()
  target_link_options(copyparty-viewer PRIVATE -static-libgcc -static-libstdc++ -Wl,--gc-sections)
  if(WIN32)
    target_link_options(copyparty-viewer PRIVATE -static -Wl,--dynamicbase -Wl,--nxcompat -Wl,--high-entropy-va)
  endif()
endif()

  enable_testing()

  target_link_libraries(core-tests PRIVATE cpv_core)
  set(core_test_arguments "")
  if(CPV_VERIFY_PUBLIC_DEFAULTS)
    list(APPEND core_test_arguments --public-defaults)
  endif()
  add_test(NAME core-tests COMMAND core-tests ${core_test_arguments})
  set_tests_properties(core-tests PROPERTIES TIMEOUT 30)
  if(CPV_TEST_HTTP)
    find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
    add_test(NAME http-fixture
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/run_http_tests.py"
        "$<TARGET_FILE:core-tests>" ${core_test_arguments})
    set_tests_properties(http-fixture PROPERTIES TIMEOUT 60)
  endif()
  add_executable(preview-tests tests/preview_tests.cpp)
  target_link_libraries(preview-tests PRIVATE cpv_preview)
  add_test(NAME preview-tests COMMAND preview-tests "${CMAKE_SOURCE_DIR}/tests/fixtures")
  set_tests_properties(preview-tests PROPERTIES TIMEOUT 30)
  add_executable(document-tests tests/document_tests.cpp)
  target_link_libraries(document-tests PRIVATE cpv_document)
  add_test(NAME document-tests COMMAND document-tests)
  add_executable(rtf-tests tests/rtf_tests.cpp)
  target_link_libraries(rtf-tests PRIVATE cpv_document)
  add_test(NAME rtf-tests COMMAND rtf-tests)
  add_executable(csv-tests tests/csv_tests.cpp)
  target_link_libraries(csv-tests PRIVATE cpv_document)
  add_test(NAME csv-tests COMMAND csv-tests)
  add_executable(text-encoding-tests tests/text_encoding_tests.cpp)
  target_link_libraries(text-encoding-tests PRIVATE cpv_preview)
  add_test(NAME text-encoding-tests COMMAND text-encoding-tests)
  set_tests_properties(document-tests rtf-tests csv-tests text-encoding-tests PROPERTIES TIMEOUT 30)
  # Display checks remain opt-in for ordinary headless developer builds.
  if(CPV_TEST_RETRO)
    add_executable(retro-tests tests/retro_tests.cpp src/retro.cpp)
    add_test(NAME retro-tests COMMAND retro-tests)

    add_executable(retro-tests EXCLUDE_FROM_ALL tests/retro_tests.cpp src/retro.cpp)

  target_link_libraries(retro-tests PRIVATE cpv_preview)
  if(MINGW)
    target_link_options(core-tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(preview-tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(document-tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(rtf-tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(csv-tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(text-encoding-tests PRIVATE -static -static-libgcc -static-libstdc++)
    target_link_options(retro-tests PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
endif()
